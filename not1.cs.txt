using System;
using System.Security.Cryptography;
using System.Text;

namespace TuProyectoAula
{
    public partial class FormPrincipal : Form
    {
        // ... Tus eventos de diseño, botones redondos, etc. ...

        private void btnAcceder_Click(object sender, EventArgs e)
        {
            string userRaw = txtUsuario.Text;
            string passRaw = txtPasswd.Text;

            // 1. Convertimos el usuario en un identificador hash invisible
            string usuarioAnonimo = BunkerCripto.GenerarIdentificadorUsuario(userRaw);

            // 2. Convertimos la contraseña en la sopa alfanumérica criptográfica
            string passwordHash = BunkerCripto.InyectarEsteroidesPassword(passRaw);

            // 3. Tu lógica All-In-One de entorno conmutado
            bool accesoConcedido = false;
            if (ENTORNO_DEBUG)
            {
                // Consulta parametrizada a tu local.db usando las cadenas hasheadas
                accesoConcedido = ValidarEnBaseDatosLocal(usuarioAnonimo, passwordHash);
            }
            else
            {
                // Consulta parametrizada hacia tu VPS remoto
                accesoConcedido = ValidarEnBaseDatosRemota(usuarioAnonimo, passwordHash);
            }

            if (accesoConcedido) {
                MessageBox.Show("Acceso autorizado. ¡Infraestructura blindada!");
                // Abrir siguiente módulo...
            } else {
                MessageBox.Show("Credenciales inválidas.");
            }
        }

        // ========================================================
        // EL BÚNKER EN C#: CLASE PRIVADA CRIPTOGRÁFICA
        // ========================================================
        private static class BunkerCripto
        {
            // Una sal estática o llave (pepper) interna para que el hash sea único
            private const string PEPPER_SECRETO = "SFMCUT_Sovereign_Salt_2026";

            public static string GenerarIdentificadorUsuario(string usuario)
            {
                using (SHA256 sha256 = SHA256.Create())
                {
                    // Combinamos el usuario con el pepper para evitar tablas de arcoíris
                    byte[] inputBytes = Encoding.UTF8.GetBytes(usuario + PEPPER_SECRETO);
                    byte[] hashBytes = sha256.ComputeHash(inputBytes);
                    
                    return BitConverter.ToString(hashBytes).Replace("-", "").ToLower();
                }
            }

            public static string InyectarEsteroidesPassword(string password)
            {
                // Usamos PBKDF2 (Rfc2898DeriveBytes) con SHA-256 para triturar la contraseña
                byte[] salt = Encoding.UTF8.GetBytes(PEPPER_SECRETO);
                
                // 50,000 iteraciones para ralentizar cualquier ataque por fuerza bruta externo
                using (var pbkdf2 = new Rfc2898DeriveBytes(password, salt, 50000, HashName.SHA256))
                {
                    byte[] hashBytes = pbkdf2.GetBytes(32); // Genera 256 bits de sopa alfanumérica
                    return Convert.ToBase64String(hashBytes); // Lo exportamos como texto limpio para la DB
                }
            }
        }
    }
}
